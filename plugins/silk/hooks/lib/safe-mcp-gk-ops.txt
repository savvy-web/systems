# (mcp__gitkraken__<op>, mcp__GitKraken__<op>, or legacy mcp__gk__<op>).

# === Write ===
# NOTE: `git_add_or_commit` and `git_push` are deliberately NOT auto-allowed.
# Commit-message validation runs on the Bash cold path only, so auto-allowing the
# MCP commit equivalent would silently bypass it. `git_push` stays prompt-gated
# as plain conservatism about a remote-mutating op — no hook blocks a push for a
# missing changeset any more (that is the human's call, enforced in CI), so this
# is a permissions choice, not a guard-bypass concern.
git_checkout
git_pull
git_stash
git_worktree
# NOTE: `pull_request_create` is auto-allowed without commit-message validation,
# mirroring the GitHub list's `create_pull_request` treatment: MCP-driven PR
# creation is a coarse-grained operation, not a commit event.
pull_request_create
pull_request_create_review
issues_add_comment
gitlens_commit_composer
