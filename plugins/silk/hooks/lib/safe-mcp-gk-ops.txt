# NOTE: `git_add_or_commit` and `git_push` are deliberately NOT auto-allowed.
# The plugin's commit-message validation and changeset-push-guard run on the
# Bash cold path only; auto-allowing the MCP equivalents would silently bypass
# both guards. Keep them prompt-gated unless equivalent MCP handlers exist.
git_checkout
git_pull
git_stash
git_worktree
# NOTE: `pull_request_create` is auto-allowed without commit-message validation,
# mirroring the GitHub list's `create_pull_request` treatment: MCP-driven PR
# creation is a coarse-grained operation, not a commit event.
pull_request_create
pull_request_create_review
issues_add_comment
gitlens_commit_composer
